# Identify the sources, rather than the build time, so host previews and the
# cross-compiled app show the same ID. Reconfigure when any input changes.
file(GLOB_RECURSE identity_files CONFIGURE_DEPENDS RELATIVE "${CMAKE_SOURCE_DIR}"
  src/* input/* contrib/* scripts/* assets/*)
list(APPEND identity_files CMakeLists.txt dependencies.lock.json
  frame-advanced-settings.vrmanifest install.sh)
list(SORT identity_files)
set(identity "")
foreach(path IN LISTS identity_files)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/${path}")
  file(SHA256 "${CMAKE_SOURCE_DIR}/${path}" digest)
  string(APPEND identity "${path}:${digest}\n")
endforeach()
string(SHA256 source_hash "${identity}")
string(SUBSTRING "${source_hash}" 0 12 FRAME_BUILD_ID)
configure_file(src/build_version.h.in generated/build_version.h @ONLY)
file(WRITE "${CMAKE_BINARY_DIR}/build-id.txt" "${FRAME_APP_VERSION}+${FRAME_BUILD_ID}\n")
message(STATUS "Frame Advanced Settings ${FRAME_APP_VERSION}+${FRAME_BUILD_ID}")

if(FRAME_BUILD_APP)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "App targets Linux. Use FRAME_BUILD_APP=OFF for host tests.")
  endif()
  set(OPENVR_LIBRARY "${CMAKE_SOURCE_DIR}/vendor/openvr/lib/libopenvr_api.so" CACHE FILEPATH "ARM64 OpenVR loader")
  # Valve's prebuilt ARM64 loader has no SONAME. Load the bundled copy explicitly,
  # avoiding build-machine paths in DT_NEEDED when cross-linking with Zig.
  add_library(openvr-loader STATIC src/openvr_dispatch.cpp)
  target_include_directories(openvr-loader PUBLIC src vendor/openvr/include)
  target_link_libraries(openvr-loader PRIVATE ${CMAKE_DL_LIBS})
  # Give our forwarding functions distinct symbols so they cannot interpose on
  # calls made internally by the real loader. These definitions apply to all users
  # of OpenVR's inline C++ interface helpers, including the Vulkan uploader.
  foreach(entry IN ITEMS VR_InitInternal2 VR_ShutdownInternal VR_GetGenericInterface
      VR_IsInterfaceVersionValid VR_GetInitToken VR_GetVRInitErrorAsEnglishDescription)
    target_compile_definitions(openvr-loader PUBLIC "${entry}=Frame_${entry}")
  endforeach()
  add_executable(frame-advanced-settings src/main.cpp src/vulkan_dispatch.cpp src/vk_texture.cpp)
  target_include_directories(frame-advanced-settings PRIVATE src vendor/vulkan/include)
  target_link_libraries(frame-advanced-settings PRIVATE drag-core drag-panel openvr-loader session-log launch-menu app-icons microphone ${CMAKE_DL_LIBS})
  target_compile_options(frame-advanced-settings PRIVATE -Wall -Wextra)
  set_target_properties(frame-advanced-settings PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
  install(TARGETS frame-advanced-settings RUNTIME DESTINATION .)
  install(FILES "${CMAKE_BINARY_DIR}/build-id.txt" DESTINATION .)
  install(FILES "${OPENVR_LIBRARY}" DESTINATION lib)
  install(FILES frame-advanced-settings.vrmanifest DESTINATION .)
  install(PROGRAMS scripts/export-logs.sh scripts/collect-logs.sh scripts/microphone-setup.sh DESTINATION .)
  install(DIRECTORY input DESTINATION .)
  install(DIRECTORY assets/icons DESTINATION assets)
  install(FILES vendor/inter/Inter-Regular.ttf vendor/inter/Inter-SemiBold.ttf DESTINATION assets/fonts)
endif()
